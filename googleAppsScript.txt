 * 
 * This script will automatically create a Google Sheet to store visitor data
 */

function doPost(e) {
  try {
    // Parse the incoming data - handle both JSON and form-encoded
    var data;
    if (e.postData && e.postData.contents) {
      try {
        data = JSON.parse(e.postData.contents);
      } catch (parseError) {
        // Try parsing as form data
        var params = e.parameter;
        if (params.data) {
          data = JSON.parse(params.data);
        } else {
          data = params;
        }
      }
    } else {
      data = e.parameter;
    }
    
    // Get or create the spreadsheet
    var ss = getOrCreateSpreadsheet();
    var sheet = ss.getActiveSheet();
    
    // Check if headers exist, if not create them
    if (sheet.getLastRow() === 0) {
      sheet.appendRow([
        'Timestamp',
        'Local Time',
        'Timezone',
        'Browser',
        'OS',
        'Device Type',
        'Screen Width',
        'Screen Height',
        'Language',
        'Platform',
        'Referrer',
        'Page URL',
        'User Agent'
      ]);
      
      // Format header row
      var headerRange = sheet.getRange(1, 1, 1, 13);
      headerRange.setFontWeight('bold');
      headerRange.setBackground('#667eea');
      headerRange.setFontColor('#ffffff');
    }
    
    // Append the data
    sheet.appendRow([
      data.timestamp || '',
      data.localTime || '',
      data.timezone || '',
      data.browser || '',
      data.os || '',
      data.deviceType || '',
      data.screenWidth || '',
      data.screenHeight || '',
      data.language || '',
      data.platform || '',
      data.referrer || '',
      data.pageUrl || '',
      data.userAgent || ''
    ]);
    
    // Auto-resize columns
    sheet.autoResizeColumns(1, 13);
    
    return ContentService.createTextOutput(JSON.stringify({
      'status': 'success',
      'message': 'Data recorded'
    })).setMimeType(ContentService.MimeType.JSON);
    
  } catch (error) {
    return ContentService.createTextOutput(JSON.stringify({
      'status': 'error',
      'message': error.toString()
    })).setMimeType(ContentService.MimeType.JSON);
  }
}

function getOrCreateSpreadsheet() {
  var sheetName = 'Link Tracker Data';
  var files = DriveApp.getFilesByName(sheetName);
  
  if (files.hasNext()) {
    // Spreadsheet exists, open it
    var file = files.next();
    return SpreadsheetApp.open(file);
  } else {
    // Create new spreadsheet
    var ss = SpreadsheetApp.create(sheetName);
    return ss;
  }
}
